cmake_minimum_required(VERSION 3.22)
project(image_squoosher_avif LANGUAGES C CXX)
include(FetchContent)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(AVIF_CODEC_AOM LOCAL CACHE STRING "" FORCE)
set(AVIF_CODEC_AOM_ENCODE OFF CACHE BOOL "" FORCE)
set(AVIF_LIBYUV OFF CACHE STRING "" FORCE)
set(AVIF_BUILD_APPS OFF CACHE BOOL "" FORCE)
set(ENABLE_NASM OFF CACHE BOOL "" FORCE)
set(ENABLE_ASM OFF CACHE BOOL "" FORCE)
FetchContent_Declare(libavif GIT_REPOSITORY https://github.com/AOMediaCodec/libavif.git GIT_TAG v1.3.0 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(libavif)
add_executable(squoosher_avif decoder.cpp)
target_compile_features(squoosher_avif PRIVATE cxx_std_17)
target_link_libraries(squoosher_avif PRIVATE avif)
